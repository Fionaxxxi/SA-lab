cmake_minimum_required(VERSION 3.22)
project(interview_tokenizer)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(FetchContent)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
FetchContent_Declare(sentencepiece GIT_REPOSITORY https://github.com/google/sentencepiece.git GIT_TAG v0.2.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(sentencepiece)
add_library(interview_tokenizer SHARED tokenizer.cpp)
target_include_directories(interview_tokenizer PRIVATE ${sentencepiece_SOURCE_DIR}/src ${sentencepiece_BINARY_DIR})
target_link_libraries(interview_tokenizer sentencepiece-static)
target_compile_features(interview_tokenizer PRIVATE cxx_std_17)
